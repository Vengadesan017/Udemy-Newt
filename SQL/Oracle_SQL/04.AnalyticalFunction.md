# Analytical functions
- use live sql https://livesql.oracle.com/landing/
  - Analytic Functions: Databases for Developers
  - function ( use with the aggregate function)
    - over
    - partition by
    - order by
    - partition by + order by
    - windowing clause
    - sliding windows
    - filtering analytical function
```sql

 -- over()
select count(*) from HR.JOBS;      - 19
select count(*) over () from HR.JOBS;     -- return 19 in all 19 rows

-- partition 
select count(*) over(PARTITION by JOB_ID),e.JOB_ID
from   hr.EMPLOYEES e

-- order by ( order the count from min to max
select count(*) over(order by JOB_ID),e.JOB_ID
from  hr.EMPLOYEES e

--partition by + order by
select count(*) over(
    partition by JOB_ID
    order by JOB_ID
    ),e.JOB_ID
from   hr.EMPLOYEES e

-- window clause
-- use order by, the database adds a default windowing clause of:
-- range between unbounded preceding  and current row
select count(*) over(
    order by JOB_ID
   rows between unbounded preceding and current row
    ),e.JOB_ID
from   hr.EMPLOYEES e

-- Sliding windows
-- The current row + the previous row
-- All rows with the same weight as the current + all rows with a weight one less than the current
select count(*) over(
    order by JOB_ID
   range between unbounded preceding and current row
    ),e.JOB_ID
from   hr.EMPLOYEES e

-- filter
select JOB_ID
from   hr.EMPLOYEES e 
group  by JOB_ID
having count(3) >= 2;
```
