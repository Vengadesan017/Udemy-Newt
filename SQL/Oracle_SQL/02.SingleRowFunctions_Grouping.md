#### in where 
- select * from emp where job = upper('manager');
- select * from emp where job = initcap('manager');

- select substr('hello', 2,2) from dual;   -- substring

#### Numeric and date

#### NUll
- select ename.NVL(comm,0) from emp  -- null will replace with 0
- select ename, job, sal, NVL(to_char(comm), 'No Data Found')
from emp
- select ename, length(ename), nvl(NULLIF(to_char(length(ename)), to_char(5)), 'length is 5') from emp;
# Grouping
- select max(sal) from emp
- select sum(sal) as sum_sal from emp
- select count(*) as count from emp
- select avg(sal), job from emp group by job
- select distict job from emp
- select count(*), job from emp group BY job 
- select count(*), job from emp group BY job having count(*)
- Select job, count(*) from emp group by job, deptno
