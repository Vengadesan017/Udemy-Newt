# control Statments
## If statments
- if condition then
  - statement
- elsif condition then
  - statement
- else
  - statement
- end if
```
DECLARE
  v_number NUMBER       := 5;
BEGIN
  IF v_number < 10  THEN
    dbms_output.put_line('10');
  ELSIF v_number < 20 THEN
    dbms_output.put_line('20');
  ELSIF v_number < 30 THEN
    dbms_output.put_line('30');
  ELSE
      dbms_output.put_line('else');
  END IF;
END;

```
## Case 
- case
  - when condition then statement
  - when condition then statement
  - else 0
- end;

## Loop
1. loop
   - loop
     - condition
       - exit
   - end loop
   - or
   - loop
     - statment
     - exit when condition
   - end loop
2. while
   - while condition
     - statment
     - exit when ( optional)
   - end loop
3. for
   - for i in [ reverse ] lower_bound .. Upper_bound loop
     - statement
   - end loop
4. Nested loop with lable
```sql 
DECLARE
 v_inner NUMBER := 1;
BEGIN
<<outer_loop>>
 FOR v_outer IN 1..5 LOOP
  dbms_output.put_line('My outer value is : ' || v_outer );
    v_inner := 1;
    <<inner_loop>>
    LOOP
      v_inner := v_inner+1;
      dbms_output.put_line('  My inner value is : ' || v_inner );
      EXIT outer_loop WHEN v_inner * v_outer >= 16;
      EXIT WHEN v_inner * v_outer >= 15;
    END LOOP inner_loop;
 END LOOP outer_loop;
END;
```
5. goto and continoue
- goto : in condition achived and forward the execution to another line
```sql
DECLARE
  v_searched_number NUMBER  := 22;
  v_is_prime        BOOLEAN := true;
BEGIN
  FOR x IN 2..v_searched_number-1 LOOP
    IF v_searched_number MOD x = 0 THEN
      dbms_output.put_line(v_searched_number|| ' is not a prime number..');
      v_is_prime := false;
      GOTO end_point;
    END IF;
  END LOOP;
  IF v_is_prime THEN 
    dbms_output.put_line(v_searched_number|| ' is a prime number..');
  END IF;
<<end_point>>
  dbms_output.put_line('Check complete..');
END;


-- continoue
DECLARE
 v_inner NUMBER := 1;
BEGIN
 FOR v_outer IN 1..10 LOOP
  dbms_output.put_line('My outer value is : ' || v_outer );
    v_inner := 1;
    WHILE v_inner * v_outer < 15 LOOP
      v_inner := v_inner + 1;
      CONTINUE WHEN MOD(v_inner * v_outer,3) = 0;
      dbms_output.put_line('  My inner value is : ' || v_inner );
    END LOOP;
 END LOOP;
END;
